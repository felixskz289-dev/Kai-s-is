let tSec=600, tInt=null;
function renderTimer(){
  const m=Math.floor(tSec/60), s=tSec%60;
  document.getElementById('timerDisplay').textContent = m+':'+String(s).padStart(2,'0');
  document.getElementById('timerDisplay').classList.toggle('low', tSec<=60 && tInt);
}
function setTimer(min){ tSec=min*60; resetTimer(); renderTimer(); }
function toggleTimer(){
  const btn=document.getElementById('timerBtn');
  if(tInt){ clearInterval(tInt); tInt=null; btn.textContent='▶ Старт'; renderTimer(); return; }
  btn.textContent='⏸ Пауза';
  tInt=setInterval(()=>{
    tSec--; renderTimer();
    if(tSec<=0){ clearInterval(tInt); tInt=null; btn.textContent='▶ Старт'; beep(); tSec=0; renderTimer(); }
  },1000);
}

function beep(){
  try{
    const ctx=new (window.AudioContext||window.webkitAudioContext)();
    [0,0.25,0.5].forEach(t=>{
      const o=ctx.createOscillator(), g=ctx.createGain();
      o.connect(g); g.connect(ctx.destination); o.frequency.value=880;
      g.gain.setValueAtTime(0.3, ctx.currentTime+t);
      g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime+t+0.2);
      o.start(ctx.currentTime+t); o.stop(ctx.currentTime+t+0.22);
    });
  }catch(e){}
}

function buildMatch(gridId, msgId, pairs){
  const grid=document.getElementById(gridId);
  grid.innerHTML='';
  const msg=document.getElementById(msgId);
  let selL=null, done=0;
  const right=pairs.map((p,i)=>({...p, key:i})).sort(()=>Math.random()-0.5);
  const leftCol=document.createElement('div'); leftCol.className='mcol';
  const rightCol=document.createElement('div'); rightCol.className='mcol';
  pairs.forEach((p,i)=>{
    const l=document.createElement('div'); l.className='mitem'; l.textContent=p.l; l.dataset.key=i;
    l.onclick=()=>{
      if(l.classList.contains('done'))return;
      leftCol.querySelectorAll('.mitem').forEach(x=>x.classList.remove('sel'));
      l.classList.add('sel'); selL=l;
    };
    leftCol.appendChild(l);
  });
  right.forEach(r=>{
    const it=document.createElement('div'); it.className='mitem'; it.textContent=r.r; it.dataset.key=r.key;
    it.onclick=()=>{
      if(it.classList.contains('done')||!selL)return;
      if(it.dataset.key===selL.dataset.key){
        it.classList.add('done'); selL.classList.add('done'); selL.classList.remove('sel'); selL=null;
        done++;
        msg.textContent='✅ Пара найдена! Осталось: '+(pairs.length-done);
        msg.className='feedback ok';
        if(done===pairs.length){ msg.textContent='🎉 Все пары собраны!'; }
      }else{
        it.classList.add('wrong'); selL.classList.add('wrong');
        msg.textContent='❌ Не совпало, попробуйте ещё!';
        msg.className='feedback no';
        const bad=it, badL=selL;
        setTimeout(()=>{bad.classList.remove('wrong'); badL.classList.remove('wrong'); badL.classList.remove('sel');},500);
        selL=null;
      }
    };
    rightCol.appendChild(it);
  });
  grid.appendChild(leftCol); grid.appendChild(rightCol);
  msg.textContent='Найдено пар: 0 / '+pairs.length; msg.className='feedback';
}

let memState={};
function initMemory(){
  const grid=document.getElementById('memGrid');
  grid.innerHTML='';
  document.getElementById('memMsg').textContent='Пары: 0 / '+MEM_TOTAL+' · Ходов: 0';
  let cards=[];
  memWords.forEach((w,i)=>{
    cards.push({type:'ch', text:w.ch, id:i});
    cards.push({type:'py', text:w.py, id:i});
  });
  cards.sort(()=>Math.random()-0.5);
  memState={first:null, lock:false, found:0, moves:0};
  cards.forEach(c=>{
    const el=document.createElement('div'); el.className='memcard';
    el.innerHTML='<div class="meminner"><div class="memface down">字</div><div class="memface up">'+c.text+'</div></div>';
    el.onclick=()=>flipMem(el, c);
    grid.appendChild(el);
  });
}
function flipMem(el, c){
  if(memState.lock || el.classList.contains('flipped')||el.classList.contains('done')) return;
  el.classList.add('flipped');
  if(!memState.first){ memState.first={el,c}; return; }
  memState.moves++;
  const f=memState.first; memState.first=null;
  if(f.c.id===c.id && f.c.type!==c.type){
    f.el.classList.add('done'); el.classList.add('done');
    memState.found++;
    document.getElementById('memMsg').textContent='Пары: '+memState.found+' / '+MEM_TOTAL+' · Ходов: '+memState.moves+(memState.found===MEM_TOTAL?' · 🎉 Готово!':'');
  }else{
    memState.lock=true;
    document.getElementById('memMsg').textContent='Пары: '+memState.found+' / '+MEM_TOTAL+' · Ходов: '+memState.moves;
    setTimeout(()=>{ f.el.classList.remove('flipped'); el.classList.remove('flipped'); memState.lock=false; },850);
  }
}
initMemory();

function nextQuiz(){
  qIdx++;
  if(qIdx>=quiz.length){
    document.getElementById('quizBox').innerHTML='<div class="qcard" style="text-align:center;font-size:20px;font-weight:800">🎉 Все вопросы пройдены! Смотрите итоговый счёт.</div>';
    document.getElementById('quizNextBtn').style.display='none';
    document.getElementById('quizProgress').textContent='';
    return;
  }
  const item=quiz[qOrder[qIdx]];
  const box=document.getElementById('quizBox');
  box.innerHTML='';
  const card=document.createElement('div'); card.className='qcard';
  card.innerHTML='<div class="q">Вопрос '+(qIdx+1)+' / '+quiz.length+': '+item.q+'</div>';
  const order=item.a.map((_,i)=>i).sort(()=>Math.random()-0.5);
  order.forEach(ai=>{
    const b=document.createElement('button'); b.className='ans'; b.textContent=item.a[ai];
    b.onclick=()=>{
      card.classList.add('locked');
      const btns=card.querySelectorAll('.ans');
      btns[item.c].classList.add('right');
      if(ai===item.c){ card.querySelector('.q').innerHTML+=' — <span style="color:var(--green)">✅ Верно!</span>'; }
      else{ b.classList.add('wrong'); card.querySelector('.q').innerHTML+=' — <span style="color:var(--red)">❌ Неверно</span>'; }
    };
    card.appendChild(b);
  });
  box.appendChild(card);
  document.getElementById('quizProgress').textContent='Отвечает команда: '+((qIdx%2===0)?'🐼 Команда 1':'🐯 Команда 2');
}
nextQuiz();
